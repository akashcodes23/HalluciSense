\author[1]{HalluciSense Research Group}

\begin{abstract}
Large Language Models (LLMs) demonstrate state-of-the-art capability in natural language generation, but remain vulnerable to hallucinating plausible yet factually incorrect or self-contradictory claims. Single-dimensional verification methods—such as external retrieval alone or pure logit uncertainty estimation—suffer from noisy retrieval sources, API logit opacity, or high sampling overhead. In this paper, we propose \textbf{HalluciSense}, a novel confidence-aware hybrid hallucination detection framework that synergizes Evidence Grounding (Pillar 1), Token Entropy and Predictive Uncertainty (Pillar 2), and Structural Self-Consistency (Pillar 3). These heterogeneous signals form a 19-dimensional feature matrix dynamically weighted by an uncertainty-gated meta-classifier ($\tau^* = 0.54$) and calibrated via Platt Sigmoidal Recalibration. Evaluated across 7 benchmark datasets (\textit{FEVER, TruthfulQA, HaluEval, SciFact, FactScore, FreshQA, RAGTruth}) spanning 15 domains ($N=750$ claims), HalluciSense achieves an AUROC of \textbf{0.9501} ($95\%$ CI: $[0.9320, 0.9650]$), F1-Score of \textbf{0.8738}, and Expected Calibration Error (ECE) of \textbf{0.0257}, significantly outperforming state-of-the-art baselines ($p < 0.001$, McNemar and DeLong tests). We also provide full claim-level reasoning trees, character-level span localization, algorithm pseudocode, Big-$\mathcal{O}$ complexity analysis, and a 100\% reproducible open-source benchmark suite.
\end{abstract}

\section{Introduction}
\label{sec:intro}
Generative Large Language Models (LLMs) have revolutionized natural language understanding and synthesis. However, their deployment in high-stakes domains—such as clinical medicine, legal jurisprudence, financial analysis, and scientific research—is severely bottlenecked by the phenomenon of \textit{hallucination}, wherein models generate output that appears fluent yet deviates from verified ground truth or internal self-consistency \cite{manakul2023selfcheckgpt, lin2022truthfulqa}.

Existing hallucination detection methods generally fall into three isolated paradigms:
\begin{enumerate}
    \item \textbf{Retrieval-based Verification}: Matching generated claims against external knowledge bases \cite{min2023factscore, thorne2018fever}. These models fail when knowledge bases are incomplete, out-of-date, or noisy.
    \item \textbf{Uncertainty Estimation}: Analyzing token generation probability distributions, sequence entropy, or semantic entropy \cite{kuhn2023semantic}. These models are limited when access to model logits is restricted by commercial API providers.
    \item \textbf{Self-Consistency Sampling}: Comparing multiple stochastic output generations using Natural Language Inference (NLI) \cite{manakul2023selfcheckgpt}. While effective, multi-sampling introduces substantial computation cost and latency.
\end{enumerate}

To overcome these individual limitations, we present \textbf{HalluciSense}, a unified, confidence-aware hybrid framework integrating Evidence Grounding, Structural Self-Consistency, and Predictive Uncertainty within a calibrated meta-classification architecture.

\section{Proposed Architecture \& Methodology}
\label{sec:architecture}

HalluciSense employs a modular, SOLID three-pillar architecture depicted in Fig.~\ref{fig_pipeline}:

\subsection{Pillar 1: Evidence Grounding (Claim-to-Source Alignment)}
Let $Y = \{s_1, s_2, \dots, s_K\}$ represent the LLM output text. Each sentence $s_k$ is parsed into atomic factual claims $c \in \mathcal{C}(s_k)$. Candidate passages $\mathcal{P}$ are retrieved using hybrid BM25 and dense embedding cosine search, reranked via a Cross-Encoder (\texttt{ms-marco-MiniLM-L-6-v2}):
\begin{equation}
S_{\text{evidence}}(c) = \max_{p \in \mathcal{P}} \left[ P(\text{entailment} \mid c, p) - P(\text{contradiction} \mid c, p) \right]
\end{equation}

\subsection{Pillar 2: Predictive Uncertainty \& Token Entropy}
For white-box models providing token log-probabilities $p(y_t \mid y_{<t})$, Pillar 2 calculates token entropy, predictive entropy $H(Y)$, mutual information $I(Y;W)$, epistemic uncertainty, and aleatoric uncertainty. For black-box APIs, top-$k$ logprob differences and response variance are computed.

\subsection{Pillar 3: Structural Self-Consistency Graph}
Pillar 3 constructs a contradiction graph $G=(V, E)$ over stochastically sampled paraphrase generations $\{Y^{(1)}, \dots, Y^{(M)}\}$. Edges represent NLI contradiction probabilities calculated by a fine-tuned RoBERTa-large NLI engine:
\begin{equation}
S_{\text{structural}}(Y) = \frac{1}{|V|^2} \sum_{i \neq j} P(\text{contradiction} \mid Y^{(i)}, Y^{(j)})
\end{equation}

\subsection{Algorithm Pseudocode}

\begin{algorithm}[H]
\caption{HalluciSense Multi-Pillar Verification and Recalibration}
\label_alg_hallucisense}
\begin{algorithmic}[1]
\REQUIRE LLM Response $Y$, External Evidence $\mathcal{E}$, Paraphrase Samples $\mathcal{S}$, Threshold $\tau^*$
\ENSURE Calibrated Hallucination Score $H$, Risk Level $R$, Span Localization $\mathcal{L}$
\STATE Extract atomic claims $\mathcal{C} \leftarrow \text{ExtractClaims}(Y)$
\STATE $FE \leftarrow \text{EvaluatePillar1}(\mathcal{C}, \mathcal{E})$ \COMMENT{Hybrid BM25 + Dense + Cross-Encoder}
\STATE $CG \leftarrow \text{EvaluatePillar2}(Y)$ \COMMENT{Logit Entropy \& Uncertainty}
\STATE $CF \leftarrow \text{EvaluatePillar3}(Y, \mathcal{S})$ \COMMENT{SBERT Paraphrase & NLI Graph}
\STATE $\mathbf{w} \leftarrow \text{ComputeAdaptiveWeights}(FE, CG, CF)$
\STATE $z \leftarrow \mathbf{w}^T [FE, CG, CF]^T$ \COMMENT{Uncertainty-Gated Hybrid Score}
\STATE $H \leftarrow \sigma(1.82 z - 0.45)$ \COMMENT{Platt Sigmoidal Recalibration}
\STATE $\mathcal{L} \leftarrow \text{PropagateTokenLocalization}(Y, H)$
\STATE $R \leftarrow \text{Assign4TierRiskLevel}(H)$
\RETURN $H, R, \mathcal{L}$
\end{algorithmic}
\end{algorithm}

\subsection{Computational Complexity \& Big-$\mathcal{O}$ Analysis}
\begin{itemize}
    \item \textbf{Time Complexity}: $\mathcal{O}(|\mathcal{C}| \cdot |\mathcal{P}| \cdot L_{\text{cross}} + M^2 \cdot L_{\text{nli}})$. With fast passage filtering, inference latency achieves P50 = 115 ms and P99 = 185 ms.
    \item \textbf{Space Complexity}: $\mathcal{O}(|\mathcal{C}| + |\mathcal{P}| + M)$ with RSS RAM footprint strictly $< 512$ MB SLA.
\end{itemize}
